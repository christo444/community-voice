\section{SYSTEM REQUIREMENTS}

\subsection{Functional Requirements}

The system supports multiple functionalities including authentication, scheme processing, administrative control, and case management.

\subsubsection{Authentication \& Profile (FR1--FR10)}
\begin{itemize}
    \item \textbf{FR1: User Registration} -- Enables new users to create accounts using phone number and PIN.
    \item \textbf{FR2: User Login} -- Allows secure login using phone number and PIN.
    \item \textbf{FR3: Auto-login} -- Maintains session for seamless re-entry.
    \item \textbf{FR4: Logout} -- Clears session and exits securely.
    \item \textbf{FR5: PIN Reset} -- Supports recovery of forgotten PIN.
    \item \textbf{FR6: Verification} -- Uses DOB for identity confirmation.
    \item \textbf{FR7: Aadhaar OCR} -- Extracts identity data using OCR scanning.
    \item \textbf{FR8: Data Merge} -- Combines front and back Aadhaar data.
    \item \textbf{FR9: Profile Editing} -- Allows correction before final submission.
    \item \textbf{FR10: Data Storage} -- Saves profile using database upsert.
\end{itemize}

\subsubsection{Interview \& Scheme (FR11--FR19)}
\begin{itemize}
    \item \textbf{FR11: Interview Capture} -- Collects eligibility-related user data.
    \item \textbf{FR12: Speech Input} -- Enables voice-based responses.
    \item \textbf{FR13: Scheme Matching} -- Retrieves relevant schemes.
    \item \textbf{FR14: Scheme Listing} -- Displays schemes in card format.
    \item \textbf{FR15: Scheme Details} -- Shows complete scheme information.
    \item \textbf{FR16: Summarization} -- Simplifies complex descriptions.
    \item \textbf{FR17: Text-to-Speech} -- Converts text into audio.
    \item \textbf{FR18: Language Toggle} -- Supports English and Malayalam.
    \item \textbf{FR19: Translation} -- Provides real-time multilingual content.
\end{itemize}

\subsubsection{Admin \& Scheme Management (FR20--FR26)}
\begin{itemize}
    \item \textbf{FR20: Admin Login} -- Secures dashboard access.
    \item \textbf{FR21: Password Update} -- Allows credential changes.
    \item \textbf{FR22: Admin Control} -- Manages admin accounts.
    \item \textbf{FR23: PDF Upload} -- Ingests schemes from PDFs.
    \item \textbf{FR24: URL Extraction} -- Extracts schemes from links.
    \item \textbf{FR25: Dashboard View} -- Displays all schemes.
    \item \textbf{FR26: Delete Scheme} -- Removes invalid entries.
\end{itemize}

\subsubsection{Paralegal \& Case Management (FR27--FR36)}
\begin{itemize}
    \item \textbf{FR27: Application Submission} -- Allows paralegal applications.
    \item \textbf{FR28: Application Review} -- Admin views requests.
    \item \textbf{FR29: Approval} -- Approves and assigns credentials.
    \item \textbf{FR30: Rejection} -- Rejects with optional reason.
    \item \textbf{FR31: Password Reset} -- Enforces first login change.
    \item \textbf{FR32: Manage Paralegals} -- Controls activation status.
    \item \textbf{FR33: Case Assignment} -- Assigns users to paralegals.
    \item \textbf{FR34: Case Tracking} -- Tracks case progress.
    \item \textbf{FR35: Case Analytics} -- Supports reassignment and insights.
    \item \textbf{FR36: Access Control} -- Restricts routes securely.
\end{itemize}

\subsection{Software Requirements}

\subsubsection{Backend Services}
\begin{itemize}
    \item \textbf{Operating System}: Linux (Ubuntu 20.04 LTS or later) for production servers; macOS or Windows with WSL2 for development.
    \item \textbf{Python}: Version 3.8 or higher for Flask applications and dependency compatibility.
    \item \textbf{Database Server}: PostgreSQL 13.0 or higher (managed via Supabase).
    \item \textbf{Web Server}: Gunicorn 20.0+ for WSGI application serving with multiple worker processes.
    \item \textbf{API Gateway}: Nginx 1.20+ for reverse proxy, SSL termination, and load balancing (optional).
    \item \textbf{Task Queue}: Celery (optional) for asynchronous email sending and AI API calls.
    \item \textbf{Message Broker}: Redis 6.0+ (optional, for caching and Celery backend).
\end{itemize}

\subsubsection{Mobile Application}
\begin{itemize}
    \item \textbf{Android}: Minimum API level 21 (Android 5.0); Target API level 33 or higher recommended.
    \item \textbf{iOS}: Minimum iOS 11.0; iOS 13.0 or higher recommended for optimal performance.

    \item \textbf{Dart}: Version 2.17 or higher (included with Flutter).
    \item \textbf{Development Tools}: Android Studio 2021.2+ or Xcode 13.0+ for compilation and testing.
\end{itemize}

\subsubsection{Web Applications}
\begin{itemize}
    \item \textbf{Node.js}: Version 16.0 or higher for build tools and runtime.
    \item \textbf{npm}: Version 7.0+ or Yarn 1.22+ for dependency management.
    \item \textbf{Web Browsers}: Chrome 90+, Firefox 88+, Safari 14+, Edge 90+.
    \item \textbf{Build Tool}: Vite 3.0 or higher for optimized frontend builds.
\end{itemize}

\subsubsection{Cloud \& Infrastructure}
\begin{itemize}

    \item \textbf{Container Orchestration}: Kubernetes 1.21+ (optional, for production scaling).

\end{itemize}

\subsection{Operational Requirements}

\subsubsection{Deployment Environment}
\begin{itemize}
    \item \textbf{Staging Environment}: Pre-production environment mirroring production setup for testing updates.
    \item \textbf{Development Environment}: Local machine or shared development server for feature development.
    \item \textbf{Testing Environment}: Separate database for QA and UAT with realistic data volumes.
    \item \textbf{Production Environment}: Isolated, secured environment with restricted access and audit logging.
\end{itemize}

\subsubsection{Backup \& Recovery}
\begin{itemize}
    \item \textbf{Database Backups}: Daily automated backups with 30-day retention (managed by Supabase).
    \item \textbf{Backup Frequency}: Real-time continuous replication; point-in-time recovery to any timestamp in last 7 days.
    \item \textbf{File Backups}: Daily backups of uploaded PDF schemes and user documents.
    \item \textbf{Recovery Time Objective (RTO)}: Maximum 1 hour to restore full system functionality.
    \item \textbf{Recovery Point Objective (RPO)}: Maximum 1 day of data loss acceptable.
    \item \textbf{Backup Testing}: Weekly test restores to validate backup integrity.
\end{itemize}

\subsubsection{Monitoring \& Alerting}
\begin{itemize}
    \item \textbf{System Monitoring}: CPU, memory, disk usage alerts when thresholds exceeded.
    \item \textbf{API Monitoring}: Response time tracking; alerts for $>$ 5 second responses or error rates $>$ 1\%.
    \item \textbf{Database Monitoring}: Query performance tracking; alerts for slow queries ($>$ 2 seconds).
    \item \textbf{Error Tracking}: Real-time notification of application exceptions and crashes.
    \item \textbf{Log Aggregation}: Centralized logging of all system events for audit trails.
    \item \textbf{Alerting Channels}: Email, SMS, and Slack notifications for critical issues.
\end{itemize}

\subsubsection{Maintenance Windows}
\begin{itemize}
    \item \textbf{Planned Downtime}: Sunday 2 AM--4 AM IST for system updates and maintenance.
    \item \textbf{Maintenance Duration}: Maximum 2 hours; advance notification to users.
    \item \textbf{Emergency Maintenance}: Zero-downtime updates using blue-green deployment strategy (recommended).
\end{itemize}

\subsection{System Constraints}

\subsubsection{Technical Constraints}
\begin{itemize}
    \item \textbf{Language}: Development restricted to Python (backend), Dart (mobile), and JavaScript (web) for consistency.
    \item \textbf{Database}: PostgreSQL only; no NoSQL databases for consistency and ACID compliance.
    \item \textbf{API Rate Limiting}: 
    \begin{enumerate}
        \item Public endpoints: 100 requests/minute per IP
        \item Authenticated endpoints: 1000 requests/minute per user
        \item File upload: 10 uploads/hour per user
    \end{enumerate}
    \item \textbf{File Size Limits}:
    \begin{enumerate}
        \item PDF schemes: Maximum 50 MB per file
        \item User documents: Maximum 20 MB per file
        \item Total storage per user: 500 MB
    \end{enumerate}
    \item \textbf{Concurrent Users}: System designed for 1000 concurrent users; load testing required for higher volumes.
\end{itemize}

\subsubsection{Performance Constraints}
\begin{itemize}
    \item \textbf{API Response Time}: Maximum 5 seconds for 95\% of requests; 10 seconds for complex operations.
    \item \textbf{Mobile App Launch Time}: Cold start $<$ 3 seconds; warm start $<$ 1 second.
    \item \textbf{Database Query Time}: 95\% of queries complete within 500ms; 99\% within 2 seconds.
    \item \textbf{OCR Processing}: On-device OCR within 3 seconds; server-side OCR within 10 seconds.
    \item \textbf{Scheme Matching}: AI matching completes within 15 seconds for average profile.
\end{itemize}

\subsubsection{Storage Constraints}
\begin{itemize}
    \item \textbf{Database Growth}: Estimated 50 MB/month for first 1 year; scaling up as user base grows.
    \item \textbf{File Storage}: PDF schemes occupy $\sim$ 5-10 GB; user documents add variable storage.
    \item \textbf{Backup Storage}: Requires 1.5x database size for daily backups.
    \item \textbf{Archive Policy}: User documents archived/deleted after 2 years; schemes retained indefinitely.
\end{itemize}

\subsubsection{Scalability Constraints}
\begin{itemize}
    \item \textbf{Vertical Scaling Limits}: Single database server limited to $\sim$ 10,000 concurrent users; horizontal scaling required beyond.
    \item \textbf{API Server Scaling}: Stateless design enables horizontal auto-scaling with load balancer.
    \item \textbf{Mobile App Updates}: Over-the-air updates from app stores; minimum 24 hours for global rollout.
    \item \textbf{Cost Scaling}: Cloud infrastructure costs increase linearly with usage; budget planning required.
\end{itemize}
